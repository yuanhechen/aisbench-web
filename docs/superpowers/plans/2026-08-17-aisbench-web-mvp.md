# AISBench Web MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a pip-installable `aisbench-web` package that starts a polished multi-user Web interface for the AISBench CLI with private models/jobs, shared datasets, a shared FIFO queue, live logs, and result comparison.

**Architecture:** A FastAPI process serves a prebuilt React application and owns an embedded SQLite database plus one in-process FIFO worker. The worker launches the official `ais_bench` executable as a process group with a generated per-job config, writes artifacts under `~/.aisbench-web/jobs`, and exposes state through REST and WebSocket APIs without modifying AISBench.

**Tech Stack:** Python 3.10+, FastAPI, Uvicorn, built-in `sqlite3`, Argon2id, Fernet encryption, pytest, React, TypeScript, Vite, Vitest, Testing Library, Playwright, plain CSS, Lucide icons.

---

## Scope and delivery strategy

This plan produces one working MVP in four vertical increments:

1. installable Python service with authentication and private model endpoints;
2. shared datasets plus a fake-runner-backed task queue;
3. real AISBench config generation, execution, logs, results, and comparison;
4. Codex-style React UI, packaged static assets, wheel smoke test, and deployment to the provided server.

Each task ends in a green test suite and a commit. Do not begin the React application until the API contracts and fake runner are stable.

```python
# src/aisbench_web/settings.py
from dataclasses import dataclass
from pathlib import Path
import os
import shutil


def discover_ais_bench() -> Path:
    override = os.environ.get("AISBENCH_WEB_AIS_BENCH_PATH")
    if override:
        candidate = Path(override).expanduser()
        if not candidate.is_file() or not os.access(candidate, os.X_OK):
            raise RuntimeError("AISBENCH_WEB_AIS_BENCH_PATH is not an executable file")
        return candidate.resolve()
    candidate = shutil.which("ais_bench")
    if candidate is None:
        raise RuntimeError(
            "ais_bench was not found in PATH; activate the working AISBench environment first"
        )
    return Path(candidate).resolve()


@dataclass(frozen=True)
class Settings:
    data_dir: Path
    ais_bench_path: Path
    max_concurrent_jobs: int

    @classmethod
    def create(cls, data_dir: Path, ais_bench_path: Path, max_concurrent_jobs: int) -> "Settings":
        if max_concurrent_jobs < 1:
            raise ValueError("max_concurrent_jobs must be at least 1")
        return cls(Path(data_dir).expanduser().resolve(), Path(ais_bench_path), max_concurrent_jobs)

    @property
    def db_path(self) -> Path:
        return self.data_dir / "aisbench-web.db"

    @property
    def secret_path(self) -> Path:
        return self.data_dir / "secret.key"

    @property
    def jobs_dir(self) -> Path:
        return self.data_dir / "jobs"

    @property
    def downloads_dir(self) -> Path:
        return self.data_dir / "downloads"

    def ensure_layout(self) -> None:
        self.data_dir.mkdir(parents=True, exist_ok=True)
        os.chmod(self.data_dir, 0o700)
        self.jobs_dir.mkdir(exist_ok=True)
        self.downloads_dir.mkdir(exist_ok=True)
```

```python
# src/aisbench_web/security.py
import hashlib
import secrets

from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError


SESSION_COOKIE = "aisbench_session"
SESSION_DAYS = 14


def hash_password(password: str) -> str:
    if len(password) < 8:
        raise ValueError("password must contain at least 8 characters")
    return PasswordHasher().hash(password)


def verify_password(encoded: str, password: str) -> bool:
    try:
        return PasswordHasher().verify(encoded, password)
    except VerifyMismatchError:
        return False


def new_session_token() -> tuple[str, str]:
    token = secrets.token_urlsafe(32)
    digest = hashlib.sha256(token.encode("utf-8")).hexdigest()
    return token, digest
```

`UserRepository` must use UUID strings, UTC ISO-8601 timestamps, parameterized SQL, case-insensitive uniqueness, and SHA-256 session-token lookup. `get_current_user` reads only the HttpOnly cookie and returns HTTP 401 for absent, expired, or revoked sessions. Registration and login set `httponly=True`, `samesite="lax"`, `path="/"`, and set `secure=True` when the request scheme is HTTPS; logout deletes both the database session and cookie. Add an Origin-check middleware for non-GET `/api/` requests: allow missing Origin for non-browser clients, but reject a present Origin whose host differs from the request host.

```python
# src/aisbench_web/jobs/states.py
from enum import Enum


class JobStatus(str, Enum):
    QUEUED = "queued"
    STARTING = "starting"
    RUNNING = "running"
    STOPPING = "stopping"
    SUCCEEDED = "succeeded"
    FAILED = "failed"
    CANCELLED = "cancelled"
    INTERRUPTED = "interrupted"


ALLOWED_TRANSITIONS = {
    JobStatus.QUEUED: {JobStatus.STARTING, JobStatus.CANCELLED},
    JobStatus.STARTING: {
        JobStatus.RUNNING,
        JobStatus.STOPPING,
        JobStatus.FAILED,
        JobStatus.INTERRUPTED,
    },
    JobStatus.RUNNING: {
        JobStatus.SUCCEEDED,
        JobStatus.FAILED,
        JobStatus.STOPPING,
        JobStatus.INTERRUPTED,
    },
    JobStatus.STOPPING: {JobStatus.CANCELLED, JobStatus.INTERRUPTED},
    JobStatus.SUCCEEDED: set(),
    JobStatus.FAILED: set(),
    JobStatus.CANCELLED: set(),
    JobStatus.INTERRUPTED: set(),
}


def require_transition(current: JobStatus, target: JobStatus) -> None:
    if target not in ALLOWED_TRANSITIONS[current]:
        raise ValueError(f"illegal job transition: {current} -> {target}")
```

`JobRepository.create()` creates the UUID, job directory paths, queued timestamps, and three immutable snapshots: parameters, the selected model endpoint, and the selected dataset manifest entry. The model snapshot stores API-key ciphertext, never plaintext; the worker decrypts it only while generating the runtime config. `claim_next()` uses `BEGIN IMMEDIATE`, selects the oldest queued job ordered by `created_at, id`, conditionally updates only when status remains queued, commits, and returns the claimed row. All user-facing reads and cancellation calls include `owner_id`. Add `queue_position(job_id)` that counts queued rows strictly ahead without returning their details.

```tsx
  server.use(
    http.get("/api/jobs/job-1", () => HttpResponse.json({
      id: "job-1", status: "running", progress: { completed: 512, total: 800 }

      requestedOffsets.push(offset);
      return HttpResponse.json({
        offset: offset === 0 ? 64 : 96,
        text: offset === 64 ? "completed request batch\n" : "startup\n",
      });

  );

  render(<JobDetailPage jobId="job-1" />);

  await waitFor(() => expect(fake.sockets).toHaveLength(1));
  fake.sockets[0].open();
  fake.sockets[0].emitJson({ type: "log", offset: 120 });
  await waitFor(() => expect(requestedOffsets).toContain(64));
  expect(await screen.findByText("completed request batch")).toBeInTheDocument();
  fake.restore();
});
```
